```vhdl
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity mini_cpu_core_top is
    generic (
        DATA_WIDTH : positive := 8
    );
    port (
        clk         : in  std_logic;
        rst         : in  std_logic;
        pm_addr_o   : out unsigned(7 downto 0);
        pm_data_i   : in  std_logic_vector(7 downto 0);
        dm_addr_o   : out unsigned(7 downto 0);
        dm_wdata_o  : out std_logic_vector(7 downto 0);
        dm_rdata_i  : in  std_logic_vector(7 downto 0);
        dm_we_o     : out std_logic;
        halted_o    : out std_logic;
        status_o    : out std_logic_vector(7 downto 0)
    );
end entity mini_cpu_core_top;

architecture rtl of mini_cpu_core_top is

    -- Define the instruction set architecture (ISA) for this CPU
    -- This example uses a simple custom 8-bit ISA with the following instructions:
    -- 0x00: NOP
    -- 0x1X: LOAD Rn, [addr]
    -- 0x2X: STORE Rn, [addr]
    -- 0x3X: ADD Rn, Rm
    -- 0x4X: SUB Rn, Rm
    -- 0x5X: AND Rn, Rm
    -- 0x6X: OR Rn, Rm
    -- 0x7X: XOR Rn, Rm
    -- 0x8X: CMP Rn, Rm
    -- 0x9X: JMP addr
    -- 0xA0: HALT

    type reg_file_type is array (natural range <>) of std_logic_vector(DATA_WIDTH-1 downto 0);
    signal reg_file : reg_file_type(0 to 7) := (others => (others => '0'));

    signal pc          : unsigned(7 downto 0) := (others => '0');
    signal instruction : std_logic_vector(7 downto 0);

    type state_type is (FETCH, DECODE, EXECUTE);
    signal current_state : state_type := FETCH;

begin

    -- Program Memory Interface
    pm_addr_o <= pc;

    -- Data Memory Interface
    dm_we_o   <= '1' when instruction(7 downto 4) = "0010" else '0';

    -- Control Unit (FSM)
    process(clk, rst)
    begin
        if rst = '1' then
            pc          <= (others => '0');
            current_state <= FETCH;
            halted_o    <= '0';
            status_o    <= (others => '0');
        elsif rising_edge(clk) then
            case current_state is
                when FETCH =>
                    instruction <= pm_data_i;
                    pc          <= pc + 1;
                    current_state <= DECODE;

                when DECODE =>
                    current_state <= EXECUTE;

                when EXECUTE =>
                    case instruction(7 downto 4) is
                        when "0000" => -- NOP
                            null;

                        when "0001" => -- LOAD Rn, [addr]
                            reg_file(to_integer(unsigned(instruction(3 downto 0)))) <= dm_rdata_i;

                        when "0010" => -- STORE Rn, [addr]
                            dm_addr_o <= unsigned(reg_file(to_integer(unsigned(instruction(3 downto 0)))));

                        when "0011" => -- ADD Rn, Rm
                            reg_file(to_integer(unsigned(instruction(3 downto 2) & '0'))) <= std_logic_vector(
                                unsigned(reg_file(to_integer(unsigned(instruction(3 downto 2) & '0')))) +
                                unsigned(reg_file(to_integer(unsigned(instruction(1 downto 0))))));

                        when "0100" => -- SUB Rn, Rm
                            reg_file(to_integer(unsigned(instruction(3 downto 2) & '0'))) <= std_logic_vector(
                                unsigned(reg_file(to_integer(unsigned(instruction(3 downto 2) & '0')))) -
                                unsigned(reg_file(to_integer(unsigned(instruction(1 downto 0))))));

                        when "0101" => -- AND Rn, Rm
                            reg_file(to_integer(unsigned(instruction(3 downto 2) & '0'))) <=
                                reg_file(to_integer(unsigned(instruction(3 downto 2) & '0'))) and
                                reg_file(to_integer(unsigned(instruction(1 downto 0))));

                        when "0110" => -- OR Rn, Rm
                            reg_file(to_integer(unsigned(instruction(3 downto 2) & '0'))) <=
                                reg_file(to_integer(unsigned(instruction(3 downto 2) & '0'))) or
                                reg_file(to_integer(unsigned(instruction(1 downto 0))));

                        when "0111" => -- XOR Rn, Rm
                            reg_file(to_integer(unsigned(instruction(3 downto 2) & '0'))) <=
                                reg_file(to_integer(unsigned(instruction(3 downto 2) & '0'))) xor
                                reg_file(to_integer(unsigned(instruction(1 downto 0))));

                        when "1000" => -- CMP Rn, Rm
                            if unsigned(reg_file(to_integer(unsigned(instruction(3 downto 2) & '0')))) =
                               unsigned(reg_file(to_integer(unsigned(instruction(1 downto 0))))) then
                                status_o <= X"FF";
                            else
                                status_o <= (others => '0');
                            end if;

                        when "1001" => -- JMP addr
                            pc <= unsigned(instruction(3 downto 0) & instruction(7 downto 4));

                        when "1010" => -- HALT
                            halted_o <= '1';

                        when others =>
                            null;
                    end case;

                    current_state <= FETCH;
            end case;
        end if;
    end process;

end architecture rtl;
```
