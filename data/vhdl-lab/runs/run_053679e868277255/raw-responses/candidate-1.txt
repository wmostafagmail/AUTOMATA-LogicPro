```vhdl
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity mini_cpu_core_top is
    generic (
        DATA_WIDTH : positive := 8
    );
    port (
        clk         : in  std_logic;
        rst         : in  std_logic;
        pm_addr_o   : out unsigned(7 downto 0);
        pm_data_i   : in  std_logic_vector(7 downto 0);
        dm_addr_o   : out unsigned(7 downto 0);
        dm_wdata_o  : out std_logic_vector(7 downto 0);
        dm_rdata_i  : in  std_logic_vector(7 downto 0);
        dm_we_o     : out std_logic;
        halted_o    : out std_logic;
        status_o    : out std_logic_vector(7 downto 0)
    );
end entity mini_cpu_core_top;

architecture rtl of mini_cpu_core_top is

    -- Define the custom 8-bit ISA
    type instruction_t is record
        opcode : std_logic_vector(3 downto 0);
        rd     : std_logic_vector(2 downto 0);
        rs1    : std_logic_vector(2 downto 0);
        rs2    : std_logic_vector(2 downto 0);
        imm    : signed(DATA_WIDTH-1 downto 0);
    end record instruction_t;

    -- Define the register file
    type reg_file_t is array (natural range <>) of std_logic_vector(DATA_WIDTH-1 downto 0);

    -- Signals for internal use
    signal pc_reg          : unsigned(7 downto 0) := (others => '0');
    signal instruction     : instruction_t;
    signal reg_file        : reg_file_t(0 to 7);
    signal alu_result      : std_logic_vector(DATA_WIDTH-1 downto 0);
    signal write_back_data : std_logic_vector(DATA_WIDTH-1 downto 0);
    signal write_back_reg  : natural range 0 to 7;
    signal write_back_en   : std_logic;

begin

    -- Instruction Fetch
    pm_addr_o <= pc_reg;

    fetch_process : process(clk, rst)
    begin
        if rst = '1' then
            pc_reg <= (others => '0');
        elsif rising_edge(clk) then
            pc_reg <= pc_reg + 1;
        end if;
    end process fetch_process;

    -- Instruction Decode
    decode_process : process(instruction)
    begin
        case instruction.opcode is
            when "0000" => -- ADD rd, rs1, rs2
                alu_result <= std_logic_vector(signed(reg_file(to_integer(unsigned(instruction.rs1))) + signed(reg_file(to_integer(unsigned(instruction.rs2)))));
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "0001" => -- SUB rd, rs1, rs2
                alu_result <= std_logic_vector(signed(reg_file(to_integer(unsigned(instruction.rs1))) - signed(reg_file(to_integer(unsigned(instruction.rs2)))));
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "0010" => -- AND rd, rs1, rs2
                alu_result <= reg_file(to_integer(unsigned(instruction.rs1))) and reg_file(to_integer(unsigned(instruction.rs2)));
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "0011" => -- OR rd, rs1, rs2
                alu_result <= reg_file(to_integer(unsigned(instruction.rs1))) or reg_file(to_integer(unsigned(instruction.rs2)));
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "0100" => -- XOR rd, rs1, rs2
                alu_result <= reg_file(to_integer(unsigned(instruction.rs1))) xor reg_file(to_integer(unsigned(instruction.rs2)));
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "0101" => -- SLT rd, rs1, rs2
                if signed(reg_file(to_integer(unsigned(instruction.rs1)))) < signed(reg_file(to_integer(unsigned(instruction.rs2)))) then
                    alu_result <= (others => '0');
                    alu_result(0) <= '1';
                else
                    alu_result <= (others => '0');
                end if;
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "0110" => -- SLL rd, rs1, imm
                alu_result <= std_logic_vector(shift_left(unsigned(reg_file(to_integer(unsigned(instruction.rs1)))), to_integer(unsigned(instruction.imm))));
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "0111" => -- SRL rd, rs1, imm
                alu_result <= std_logic_vector(shift_right(unsigned(reg_file(to_integer(unsigned(instruction.rs1)))), to_integer(unsigned(instruction.imm))));
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "1000" => -- ADDI rd, rs1, imm
                alu_result <= std_logic_vector(signed(reg_file(to_integer(unsigned(instruction.rs1))) + instruction.imm));
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "1001" => -- ANDI rd, rs1, imm
                alu_result <= reg_file(to_integer(unsigned(instruction.rs1))) and std_logic_vector(instruction.imm);
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "1010" => -- ORI rd, rs1, imm
                alu_result <= reg_file(to_integer(unsigned(instruction.rs1))) or std_logic_vector(instruction.imm);
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "1011" => -- XORI rd, rs1, imm
                alu_result <= reg_file(to_integer(unsigned(instruction.rs1))) xor std_logic_vector(instruction.imm);
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "1100" => -- SLTI rd, rs1, imm
                if signed(reg_file(to_integer(unsigned(instruction.rs1)))) < instruction.imm then
                    alu_result <= (others => '0');
                    alu_result(0) <= '1';
                else
                    alu_result <= (others => '0');
                end if;
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "1101" => -- SLLI rd, rs1, imm
                alu_result <= std_logic_vector(shift_left(unsigned(reg_file(to_integer(unsigned(instruction.rs1)))), to_integer(unsigned(instruction.imm))));
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when "1110" => -- SRLI rd, rs1, imm
                alu_result <= std_logic_vector(shift_right(unsigned(reg_file(to_integer(unsigned(instruction.rs1)))), to_integer(unsigned(instruction.imm))));
                write_back_reg <= to_integer(unsigned(instruction.rd));
                write_back_en  <= '1';
            when others =>
                alu_result     <= (others => '0');
                write_back_reg <= 0;
                write_back_en  <= '0';
        end case;
    end process decode_process;

    -- Register File
    reg_file_process : process(clk, rst)
    begin
        if rst = '1' then
            for i in 0 to 7 loop
                reg_file(i) <= (others => '0');
            end loop;
        elsif rising_edge(clk) then
            if write_back_en = '1' then
                reg_file(write_back_reg) <= write_back_data;
            end if;
        end if;
    end process reg_file_process;

    -- Write Back
    write_back_data <= alu_result;

    -- Data Memory Interface
    dm_addr_o   <= unsigned(reg_file(0));
    dm_wdata_o  <= reg_file(1);
    dm_we_o     <= '0'; -- No writes to data memory in this example

    -- Halted and Status Signals
    halted_o <= '0';
    status_o <= (others => '0');

end architecture rtl;
```
