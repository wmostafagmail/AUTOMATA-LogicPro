library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity mini_cpu_core_top is
  generic (
    DATA_WIDTH : positive := 8
  );
  port (
    clk : in std_logic;
    rst : in std_logic;
    pm_addr_o : out unsigned(7 downto 0);
    pm_data_i : in std_logic_vector(7 downto 0);
    dm_addr_o : out unsigned(7 downto 0);
    dm_wdata_o : out std_logic_vector(7 downto 0);
    dm_rdata_i : in std_logic_vector(7 downto 0);
    dm_we_o : out std_logic;
    halted_o : out std_logic;
    status_o : out std_logic_vector(7 downto 0)
  );
end entity mini_cpu_core_top;

library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

architecture rtl of mini_cpu_core_top is
    signal pc : unsigned(7 downto 0) := (others => '0');
    signal ir : std_logic_vector(7 downto 0);
    signal alu_out : std_logic_vector(7 downto 0);
    signal regfile : std_logic_vector(7 downto 0) := (others => '0');
    signal halted : std_logic := '0';
begin
    process(clk, rst)
    begin
        if rst = '1' then
            pc <= (others => '0');
            halted <= '1';
        elsif rising_edge(clk) then
            if halted = '0' then
                pm_addr_o <= pc;
                pc <= pc + 1;
                ir <= pm_data_i;
                -- Decode and execute
                case ir is
                    when "00000000" => -- NOP
                        null;
                    when "00000001" => -- LDR R0, [R1]
                        regfile(0) <= dm_rdata_i;
                    when "00000010" => -- STR R0, [R1]
                        dm_addr_o <= regfile(1);
                        dm_wdata_o <= regfile(0);
                        dm_we_o <= '1';
                    when others =>
                        halted <= '1';
                end case;
            end if;
        end if;
    end process;

    status_o <= (others => '0');
    halted_o <= halted;
end architecture rtl;
