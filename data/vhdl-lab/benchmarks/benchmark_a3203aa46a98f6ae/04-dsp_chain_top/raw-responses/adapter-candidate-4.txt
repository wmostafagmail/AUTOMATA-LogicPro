library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity dsp_chain_top is
    generic (
        DATA_WIDTH : positive := 16
    );
    port (
        clk      : in  std_logic;
        rst      : in  std_logic;
        sample_i : in  signed(15 downto 0);
        sample_valid_i : in  std_logic;
        sample_ready_o : out std_logic;
        result_o : out signed(31 downto 0);
        result_valid_o : out std_logic
    );
end entity dsp_chain_top;

architecture rtl of dsp_chain_top is

    type sample_reg_type is record
        valid : std_logic;
        data  : signed(15 downto 0);
    end record;

    signal sample_reg : sample_reg_type;

    type fir_reg_type is record
        data : signed(31 downto 0);
    end record;

    signal fir_reg : fir_reg_type;

begin

    process (clk, rst) is
    begin
        if rst = '1' then
            sample_reg.valid <= '0';
            sample_reg.data  <= (others => '0');
            fir_reg.data      <= (others => '0');
        elsif rising_edge(clk) then
            sample_reg.valid <= sample_valid_i;
            sample_reg.data  <= sample_i;
            fir_reg.data      <= sample_reg.data * to_signed(1, DATA_WIDTH);
        end if;
    end process;

    result_o <= fir_reg.data;
    result_valid_o <= fir_reg.data(31);

end architecture rtl;
